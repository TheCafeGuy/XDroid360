# XDroid360 native library: upstream xenia-edge built as static libraries for
# arm64-v8a, plus the Android glue, linked into libxdroid360.so.
#
# Configured by Gradle (:xdroid-core externalNativeBuild), which supplies the
# NDK toolchain and the ABI. Not meant to be configured on its own.
cmake_minimum_required(VERSION 3.22)

# Tested = shipped: the Gradle debug variant (debuggable, for adb run-as)
# builds the exact native code of the release, so no upstream Debug-only path
# ($<CONFIG:Debug>: DEBUG) runs in tests that players never get. AGP always
# passes its own CMAKE_BUILD_TYPE, so it is overridden here.
set(CMAKE_BUILD_TYPE Release)

project(xdroid360 LANGUAGES C CXX ASM)

# cpp -> main -> src -> xdroid-core -> repository root.
get_filename_component(XDROID_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.."
                       ABSOLUTE)

# version.h normally comes from xenia-build.py; generate the same macros
# from git here. Rewritten only when the content changes, so an unchanged
# commit does not rebuild everything.
find_program(XDROID_GIT git)
set(_branch "unknown")
set(_commit "unknown")
set(_commit_short "unknown")
if(XDROID_GIT)
  execute_process(COMMAND "${XDROID_GIT}" rev-parse --abbrev-ref HEAD
                  WORKING_DIRECTORY "${XDROID_ROOT}"
                  OUTPUT_VARIABLE _branch OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET)
  execute_process(COMMAND "${XDROID_GIT}" rev-parse HEAD
                  WORKING_DIRECTORY "${XDROID_ROOT}"
                  OUTPUT_VARIABLE _commit OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET)
  execute_process(COMMAND "${XDROID_GIT}" rev-parse --short HEAD
                  WORKING_DIRECTORY "${XDROID_ROOT}"
                  OUTPUT_VARIABLE _commit_short
                  OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
set(_version_h "${CMAKE_BINARY_DIR}/version.h")
set(_version_text "// Generated by xdroid-core/src/main/cpp/CMakeLists.txt.
#ifndef GENERATED_VERSION_H_
#define GENERATED_VERSION_H_
#define XE_BUILD_BRANCH \"${_branch}\"
#define XE_BUILD_COMMIT \"${_commit}\"
#define XE_BUILD_COMMIT_SHORT \"${_commit_short}\"
#define XE_BUILD_DATE __DATE__
#endif  // GENERATED_VERSION_H_
")
set(_version_old "")
if(EXISTS "${_version_h}")
  file(READ "${_version_h}" _version_old)
endif()
if(NOT _version_old STREQUAL _version_text)
  file(WRITE "${_version_h}" "${_version_text}")
endif()

set(XENIA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(XENIA_BUILD_MISC OFF CACHE BOOL "" FORCE)
# The upstream LTO switch only covers the Release config; XDROID_LTO below
# covers every config.
set(XENIA_ENABLE_LTO OFF CACHE BOOL "" FORCE)

# ThinLTO over upstream and glue. The archiver must be the NDK's llvm-ar so
# the bitcode matches the compiler.
option(XDROID_LTO "ThinLTO for every config" OFF)
if(XDROID_LTO)
  if(NOT LLVM_AR OR NOT LLVM_RANLIB)
    message(FATAL_ERROR "XDROID_LTO needs -DLLVM_AR and -DLLVM_RANLIB")
  endif()
  set(CMAKE_AR "${LLVM_AR}" CACHE FILEPATH "" FORCE)
  set(CMAKE_RANLIB "${LLVM_RANLIB}" CACHE FILEPATH "" FORCE)
  add_compile_options(-flto=thin)
  add_link_options(-flto=thin -fuse-ld=lld)
endif()

add_subdirectory("${XDROID_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/upstream")

# adrenotools: static library linked in. Its hook libraries are shared and
# must be packaged next to libxdroid360.so, where adrenotools looks for them.
add_subdirectory("${XDROID_ROOT}/third_party/libadrenotools"
                 "${CMAKE_CURRENT_BINARY_DIR}/adrenotools")

add_library(xdroid360 SHARED
  xdroid_ac6_fixes.cc
  xdroid_ac6_jobs.cc
  xdroid_ac6_jobs_trace.cc
  xdroid_ac6_timing.cc
  xdroid_adpf.cc
  xdroid_app.cc
  xdroid_audio_driver.cc
  xdroid_content.cc
  xdroid_defaults.cc
  xdroid_driver.cc
  xdroid_events.cc
  xdroid_gamepad.cc
  xdroid_gpu_bench.cc
  xdroid_ib_snapshot.cc
  xdroid_jni.cc
  xdroid_latency.cc
  xdroid_live.cc
  xdroid_time_coupling.cc
)
add_dependencies(xdroid360 main_hook hook_impl)
target_compile_features(xdroid360 PRIVATE cxx_std_20 c_std_17)
# GPU-BENCH capture builds: our sources must see the trace writer too.
if(XENIA_ENABLE_GPU_TRACE)
  target_compile_definitions(xdroid360 PRIVATE XE_OPTION_GPU_TRACE=1)
endif()
# Logged at boot as "XDroid360: build <hash>" to match a binary to a commit.
add_custom_target(xdroid_build_id
  COMMAND "${CMAKE_COMMAND}" "-DROOT=${XDROID_ROOT}"
          "-DOUT=${CMAKE_CURRENT_BINARY_DIR}/xdroid_build_id.h"
          -P "${CMAKE_CURRENT_SOURCE_DIR}/build_id.cmake"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/xdroid_build_id.h"
  VERBATIM)
add_dependencies(xdroid360 xdroid_build_id)
# The upstream include directories are set in the upstream directory scope.
target_include_directories(xdroid360 PRIVATE
  "${CMAKE_CURRENT_BINARY_DIR}"
  "${XDROID_ROOT}"
  "${XDROID_ROOT}/src"
)
target_include_directories(xdroid360 SYSTEM PRIVATE
  "${XDROID_ROOT}/third_party"
)
target_link_options(xdroid360 PRIVATE
  # 16 KB pages (Android 15+ devices).
  "-Wl,-z,max-page-size=16384"
  # Export only the JNI entry points.
  "-Wl,--exclude-libs,ALL"
)
target_link_libraries(xdroid360 PRIVATE
  adrenotools
  xenia-apu
  xenia-apu-nop
  xenia-base
  xenia-core
  xenia-cpu
  xenia-cpu-backend-a64
  xenia-gpu
  xenia-gpu-null
  xenia-gpu-vulkan
  xenia-hid
  xenia-hid-nop
  xenia-kernel
  xenia-patcher
  xenia-ui
  xenia-ui-vulkan
  xenia-vfs
  imgui
  aaudio
  android
  log
)
